Czeit.c 
// zeit.c: Prozess oder Thread, was ist schneller?
// Übersetzen: gcc -O2 zeit.c -o zeit -pthread
// Starten:    ./zeit
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/wait.h>
#include <time.h>
#include <unistd.h>

#define ANZAHL 2000                 // wie oft wir anlegen
#define MB 100                      // wie viele Daten wir austauschen
#define GROESSE ((size_t)MB << 20)

static double jetzt(void) {         // Zeit in Sekunden
    struct timespec t;
    clock_gettime(CLOCK_MONOTONIC, &t);
    return t.tv_sec + t.tv_nsec / 1e9;
}

static void *leer(void *arg) { return arg; }

static unsigned char *daten;        // gemeinsamer Speicher der Threads
static unsigned long summe;

static void *summiere(void *arg) {  // Thread liest die Daten direkt
    unsigned long s = 0;
    for (size_t i = 0; i < GROESSE; i++) s += daten[i];
    summe = s;
    return arg;
}

int main(void) {
    // 1) Anlegen: neuer Prozess gegen neuen Thread
    double t0 = jetzt();
    for (int i = 0; i < ANZAHL; i++) {
        pid_t pid = fork();
        if (pid == 0) _exit(0);     // Kind endet sofort
        waitpid(pid, NULL, 0);
    }
    double prozess = (jetzt() - t0) / ANZAHL;

    t0 = jetzt();
    for (int i = 0; i < ANZAHL; i++) {
        pthread_t t;
        pthread_create(&t, NULL, leer, NULL);
        pthread_join(t, NULL);
    }
    double thread = (jetzt() - t0) / ANZAHL;

    printf("Anlegen (Mittel aus %d)\n", ANZAHL);
    printf("  neuer Prozess: %8.1f us\n", prozess * 1e6);
    printf("  neuer Thread:  %8.1f us   -> %.0fx schneller\n\n", thread * 1e6, prozess / thread);

    // 2) Austausch: der andere soll 100 MB bekommen und aufsummieren
    daten = malloc(GROESSE);
    unsigned char *puffer = malloc(1 << 20);
    int hin[2], rueck[2];
    if (pipe(hin) || pipe(rueck)) return 1;

    pid_t pid = fork();
    if (pid == 0) {                 // Prozess B: liest alles aus der Pipe
        close(hin[1]);
        unsigned long s = 0;
        ssize_t n;
        while ((n = read(hin[0], puffer, 1 << 20)) > 0)
            for (ssize_t i = 0; i < n; i++) s += puffer[i];
        if (write(rueck[1], &s, sizeof s) < 0) _exit(1);
        _exit(0);
    }
    close(hin[0]);
    for (size_t i = 0; i < GROESSE; i++) daten[i] = (unsigned char)i;

    t0 = jetzt();                   // Prozess A schickt alles über das Betriebssystem
    for (size_t gesendet = 0; gesendet < GROESSE;) {
        ssize_t n = write(hin[1], daten + gesendet, GROESSE - gesendet);
        if (n <= 0) { perror("write"); return 1; }
        gesendet += (size_t)n;
    }
    close(hin[1]);
    unsigned long s_prozess;
    if (read(rueck[0], &s_prozess, sizeof s_prozess) != sizeof s_prozess) return 1;
    prozess = jetzt() - t0;
    waitpid(pid, NULL, 0);

    t0 = jetzt();                   // Thread 2 liest direkt aus dem gemeinsamen Speicher
    pthread_t t;
    pthread_create(&t, NULL, summiere, NULL);
    pthread_join(t, NULL);
    thread = jetzt() - t0;

    printf("Austausch von %d MB\n", MB);
    printf("  zwei Prozesse: %8.1f ms\n", prozess * 1e3);
    printf("  zwei Threads:  %8.1f ms   -> %.0fx schneller\n", thread * 1e3, prozess / thread);
    if (s_prozess != summe) printf("Fehler: Summen verschieden\n");
    return 0;
}
Probier es selbst

Nicht lesen, ausführen. Das Programm misst zwei Dinge:

  • Anlegen: 2000-mal einen Prozess starten und beenden, dann 2000-mal einen Thread.
  • Austausch: 100 MB weitergeben, an einen Prozess über eine Pipe, an einen Thread über den gemeinsamen Speicher.
  1. Code kopieren, als zeit.c speichern
  2. gcc -O2 zeit.c -o zeit -pthread
  3. ./zeit
